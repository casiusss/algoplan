       ```ts
       import type { InboxItem } from "@multica/core/types";

       export type InboxBucket = "today" | "yesterday" | "this_week" | "older";

       export interface InboxBucketGroup {
         bucket: InboxBucket;
         items: InboxItem[];
       }

       const ORDER: InboxBucket[] = ["today", "yesterday", "this_week", "older"];

       export function groupInboxByDate(items: InboxItem[], now: Date = new Date()): InboxBucketGroup[] {
         const startOfToday = new Date(now);
         startOfToday.setHours(0, 0, 0, 0);
         const startOfYesterday = new Date(startOfToday);
         startOfYesterday.setDate(startOfToday.getDate() - 1);
         const startOf7DaysAgo = new Date(startOfToday);
         startOf7DaysAgo.setDate(startOfToday.getDate() - 6); // today + 6 prior calendar days = "last 7 days"

         const buckets: Record<InboxBucket, InboxItem[]> = { today: [], yesterday: [], this_week: [], older: [] };

         for (const item of items) {
           const ts = new Date(item.created_at);
           if (ts >= startOfToday) buckets.today.push(item);
           else if (ts >= startOfYesterday) buckets.yesterday.push(item);
           else if (ts >= startOf7DaysAgo) buckets.this_week.push(item);
           else buckets.older.push(item);
         }

         return ORDER.flatMap((bucket) => buckets[bucket].length ? [{ bucket, items: buckets[bucket] }] : []);
       }
       ```
       Per UI-SPEC §Sub-Phase INB §Date-bucket grouping:
       - "Today" = today's calendar date (local).
       - "Yesterday" = yesterday's calendar date.
       - "This week" = within the last 7 days but NOT today/yesterday → days [-6, -3] inclusive (since 0 = today and -1 = yesterday).
         **Boundary clarification**: per UI-SPEC test row "6 days ago → this_week, 7 days ago → older". So "this week" is days `-6 .. -2` (5 days), then -7+ is older. Implementation uses `>= startOfYesterday` covers -1, then `>= startOf7DaysAgo` (set as `today - 6`) covers -2 .. -6, then else is -7+. Verify with the spec test fixtures.

       ```ts
       import { useEffect } from "react";

       export function useInboxShortcut(key: string, handler: () => void): void {
         useEffect(() => {
           const onKeyDown = (e: KeyboardEvent) => {
             if (e.key.toLowerCase() !== key.toLowerCase()) return;
             if (e.metaKey || e.ctrlKey || e.altKey) return;
             const active = document.activeElement;
             if (active) {
               const tag = active.tagName;
               if (tag === "INPUT" || tag === "TEXTAREA") return;
               if (active.getAttribute("contenteditable") === "true") return;
             }
             handler();
           };
           window.addEventListener("keydown", onKeyDown);
           return () => window.removeEventListener("keydown", onKeyDown);
         }, [key, handler]);
       }
       ```

    3. **`packages/core/inbox/use-inbox-filter-store.ts`**:
       ```ts
       import { create } from "zustand";
       import type { InboxItem, InboxItemType } from "../types/inbox";

       interface InboxFilterState {
         selectedTypes: Set<InboxItemType>;
         toggleType: (type: InboxItemType) => void;
         clearFilters: () => void;
       }

       export const useInboxFilterStore = create<InboxFilterState>((set) => ({
         selectedTypes: new Set(),
         toggleType: (type) =>
           set((state) => {
             const next = new Set(state.selectedTypes);
             if (next.has(type)) next.delete(type); else next.add(type);
             return { selectedTypes: next };
           }),
         clearFilters: () => set({ selectedTypes: new Set() }),
       }));

       export function applyInboxFilter(items: InboxItem[], selectedTypes: Set<InboxItemType>): InboxItem[] {
         if (selectedTypes.size === 0) return items;
         return items.filter((item) => selectedTypes.has(item.type));
       }
       ```
