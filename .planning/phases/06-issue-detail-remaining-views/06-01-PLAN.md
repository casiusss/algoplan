       ```tsx
       import { SegmentedControl, SegmentedControlItem } from "@multica/ui/components/ui/segmented-control";
       import { Button } from "@multica/ui/components/ui/button";
       import { X } from "lucide-react";
       import type { IssuePriority } from "@multica/core/types";

       const ENUM_TO_LABEL: Record<Exclude<IssuePriority, "none">, "p0" | "p1" | "p2" | "p3"> = {
         urgent: "p0", high: "p1", medium: "p2", low: "p3",
       };
       const LABEL_TO_ENUM: Record<"p0" | "p1" | "p2" | "p3", Exclude<IssuePriority, "none">> = {
         p0: "urgent", p1: "high", p2: "medium", p3: "low",
       };

       interface Props {
         value: IssuePriority;
         onChange: (priority: IssuePriority) => void;
         disabled?: boolean;
       }

       export function IssuePrioritySegmentedControl({ value, onChange, disabled }: Props) {
         const activeLabel = value === "none" ? "" : ENUM_TO_LABEL[value];
         return (
           <div className="space-y-1.5">
             <SegmentedControl
               aria-label="Priorität"
               value={activeLabel}
               onValueChange={(label) => onChange(LABEL_TO_ENUM[label as "p0" | "p1" | "p2" | "p3"])}
               disabled={disabled}
               className="w-full"
               colorByValue={{
                 p0: "text-tag-p0", p1: "text-tag-p1", p2: "text-tag-p2", p3: "text-tag-p3",
               }}
             >
               <SegmentedControlItem value="p0">P0</SegmentedControlItem>
               <SegmentedControlItem value="p1">P1</SegmentedControlItem>
               <SegmentedControlItem value="p2">P2</SegmentedControlItem>
               <SegmentedControlItem value="p3">P3</SegmentedControlItem>
             </SegmentedControl>
             {value !== "none" && (
               <button
                 type="button"
                 onClick={() => onChange("none")}
                 className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground"
                 disabled={disabled}
               >
                 <X className="size-3" /> Priorität entfernen
               </button>
             )}
           </div>
         );
       }
       ```
       Per UI-SPEC §Hard Constraints #3: mapping is fixed `urgent→P0, high→P1, medium→P2, low→P3, none→excluded`.
