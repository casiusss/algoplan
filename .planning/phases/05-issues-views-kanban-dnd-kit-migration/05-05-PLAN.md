  <action>
    Step 1 — Create the spec:
    ```typescript
    import { test, expect } from "@playwright/test";
    import { loginAsDefault, createTestApi } from "./helpers";
    import type { TestApiClient } from "./fixtures";

    let api: TestApiClient;

    test.beforeEach(async ({ page }) => {
      api = await createTestApi();
      // Seed: 15 issues in "in_progress" so its column scrolls
      for (let i = 0; i < 15; i++) {
        await api.createIssue(`Scroll seed ${i}`, { status: "in_progress" });
      }
      // 1 issue in "todo" — the source card we'll drag
      await api.createIssue("Source card", { status: "todo" });
      await loginAsDefault(page);
    });

    test.afterEach(async () => {
      await api.cleanup();
    });

    test("AutoScroller: drop in scrolled column lands where pointer indicates (KBN-02)", async ({ page }) => {
      await page.goto("/[workspace]/issues");
      await page.waitForSelector('[data-board-column-body]');

      // Find the in_progress column body and scroll it to the bottom
      const inProgressBody = page.locator('[data-board-column-body]').filter({
        has: page.locator('[data-board-column-status-label]:has-text("In Progress")'),
      });
      // Note: the data-board-column-body lives INSIDE the column wrapper that ALSO holds the header.
      // We may need to walk up to the column wrapper — adjust the locator to target the actual scrollable element.
      // Use the closest scrollable ancestor:
      await inProgressBody.evaluate((el) => {
        const scrollable = el.closest('[data-board-column-root]') ?? el;
        scrollable.scrollTo(0, scrollable.scrollHeight);
      });

      // Find the source card and a known target card near the bottom of the in_progress column
      const sourceCard = page.locator('[data-board-card-root]').filter({ hasText: 'Source card' });
      const targetCard = page.locator('[data-board-card-root]').filter({ hasText: 'Scroll seed 14' });

      // Get positions
      const targetBox = await targetCard.boundingBox();
      if (!targetBox) throw new Error("Target card not in viewport");

      // Drag source card onto the target card position
      await sourceCard.dragTo(targetCard, {
        targetPosition: { x: targetBox.width / 2, y: targetBox.height / 2 },
      });

      // Assert: source card landed in in_progress column AND adjacent to "Scroll seed 14"
      const inProgressCards = inProgressBody.locator('[data-board-card-root]');
      const titles = await inProgressCards.allInnerTexts();
      const sourceIdx = titles.findIndex((t) => t.includes('Source card'));
      const targetIdx = titles.findIndex((t) => t.includes('Scroll seed 14'));
      expect(sourceIdx).toBeGreaterThanOrEqual(0);
      // Source should be adjacent to (within 1 of) the target — NOT at index 0 (top) which would indicate scroll drift.
      expect(Math.abs(sourceIdx - targetIdx)).toBeLessThanOrEqual(1);
    });
    ```

    NOTE on Playwright drag-and-drop with @dnd-kit: Playwright's `dragTo` uses native HTML5 drag events; @dnd-kit/react uses pointer events. If `dragTo` doesn't trigger the drag, fall back to manual pointer events:

    - On context 1: complete the drop.
    - Assert: card "X" lands and STAYS in "in_progress" (not flickering back to "todo").

    import { test, expect, type Browser, type BrowserContext, type Page } from "@playwright/test";
    import { loginAsDefault, createTestApi } from "./helpers";
    import type { TestApiClient } from "./fixtures";

    let api: TestApiClient;

    test.beforeEach(async () => {
      api = await createTestApi();
      // Seed: card to drag + a different card to update from context 2
      await api.createIssue("Drag race source", { status: "todo" });
      await api.createIssue("Other card for update", { status: "todo" });
    });

    test.afterEach(async () => {
      await api.cleanup();
    });

    test("WS event mid-drag does NOT snap card back (KBN-01)", async ({ browser }) => {
      // Context 1 — the drag actor
      const ctx1 = await browser.newContext();
      const page1 = await ctx1.newPage();
      await loginAsDefault(page1);
      await page1.goto("/[workspace]/issues");
      await page1.waitForSelector('[data-board-column-body]');

      // Context 2 — the WS event trigger
      const ctx2 = await browser.newContext();
      const page2 = await ctx2.newPage();
      await loginAsDefault(page2);
      await page2.goto("/[workspace]/issues");
      await page2.waitForSelector('[data-board-column-body]');

      const sourceCard = page1.locator('[data-board-card-root]').filter({ hasText: 'Drag race source' });
      const inProgressBody = page1.locator('[data-board-column-body]').filter({
        has: page1.locator('[data-board-column-status-label]:has-text("In Progress")'),
      });

      // Start drag on page1 — manual pointer events so we can pause mid-flight
      const sourceBox = await sourceCard.boundingBox();
      const targetBox = await inProgressBody.boundingBox();
      if (!sourceBox || !targetBox) throw new Error("Boxes missing");

      await page1.mouse.move(sourceBox.x + sourceBox.width / 2, sourceBox.y + sourceBox.height / 2);
      await page1.mouse.down();
      // Move toward in_progress — but stop short of releasing
      await page1.mouse.move(targetBox.x + targetBox.width / 2, targetBox.y + targetBox.height / 2, { steps: 10 });

      // While drag is mid-flight on page1, fire a WS-triggering update on page2
      // Use the API client to update the OTHER issue (not the one being dragged) to keep concerns clean
      const otherCardOnPage2 = page2.locator('[data-board-card-root]').filter({ hasText: 'Other card for update' });
      // Trigger an update — easiest: change the title via API; this dispatches a WS event to page1
      // (createTestApi exposes updateIssue — use a no-op patch like setting title to itself + " ")
      // For determinism, do this through the API client directly instead of through page2 UI:
      // (we still opened page2 to ensure WS subscription is live, but the update goes through API)
      // Find the other issue's id via api.cleanup() bookkeeping or via DOM inspection on page2:
      const otherId = await otherCardOnPage2.getAttribute('data-issue-id'); // requires data-issue-id on board-card-root; if absent, look it up via API list
      if (otherId) {
        await api.updateIssue(otherId, { title: "Other card for update (touched)" });
      }
      // Wait a tick to let WS event arrive at page1
      await page1.waitForTimeout(150);

      // Complete the drop
      await page1.mouse.up();

      // Assert: source card is in in_progress column on page1
      await expect(
        inProgressBody.locator('[data-board-card-root]').filter({ hasText: 'Drag race source' }),
      ).toBeVisible();

      // Assert: NOT in todo column
      const todoBody = page1.locator('[data-board-column-body]').filter({
        has: page1.locator('[data-board-column-status-label]:has-text("Todo")'),
      });
      await expect(
        todoBody.locator('[data-board-card-root]').filter({ hasText: 'Drag race source' }),
      ).toHaveCount(0);

      // Assert: no error toast
      await expect(page1.getByText("Issue konnte nicht verschoben werden")).toHaveCount(0);

      await ctx1.close();
      await ctx2.close();
    });
