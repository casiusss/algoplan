Base UI v1.3 ToggleGroup type contract (verified from installed `.d.ts`, RESEARCH lines 226-239):

```typescript
interface ToggleGroupProps<Value extends string> {
  value?: readonly Value[];
  defaultValue?: readonly Value[];
  onValueChange?: (groupValue: Value[], eventDetails: ToggleGroup.ChangeEventDetails) => void;
  disabled?: boolean;
  orientation?: "horizontal" | "vertical";
  loopFocus?: boolean;            // default true → ArrowLeft from item 0 wraps to last
  multiple?: boolean;             // default false → single-select
}
```

Critical wrapper rules:
- Pass `value={[value]}` (array of one) to the primitive.
- Pass `multiple={false}` explicitly (NOT `toggleMultiple={false}`, which does not exist).
- Inside `onValueChange={(next) => …}`: `if (next.length > 0) onValueChange(next[0])` — swallow the empty-array deselect.
- Pass `aria-label` straight through to the primitive root.

Visual contract (UI-SPEC §4):
- Track: `bg-muted` + `p-[3px]` + `rounded-lg` + `inline-flex h-6 w-fit items-center gap-0`.
- Item: `h-full min-w-8 px-3 text-sm font-medium text-muted-foreground` + `rounded-md`.
- Item active (`data-pressed`): `bg-background text-foreground shadow-sm`.
- Item hover: `hover:text-foreground`.
- Item focus: `focus-visible:ring-[3px] focus-visible:ring-ring/50 focus-visible:outline-none` (per item, not on root).
- Item disabled: `disabled:pointer-events-none disabled:opacity-50`.

import { useState } from "react";

import { SegmentedControl, SegmentedControlItem } from "./segmented-control";

function Harness({
  initial = "a",
  onChange,
  ariaLabel = "Test",
  disabledValues = [] as string[],
}: {
  initial?: string;
  onChange?: (v: string) => void;
  ariaLabel?: string;
  disabledValues?: string[];

  return (
    <>
      <button type="button">before</button>
      <SegmentedControl
        value={v}
        onValueChange={(next) => {
          onChange?.(next);
          setV(next);

        aria-label={ariaLabel}

          A

          B
        </SegmentedControlItem>

```typescript
"use client"

import * as React from "react"
import { Toggle as TogglePrimitive } from "@base-ui/react/toggle"
import { ToggleGroup as ToggleGroupPrimitive } from "@base-ui/react/toggle-group"

import { cn } from "@multica/ui/lib/utils"

/**
 * SegmentedControl — single-select value picker styled like a TabsList track
 * with a lifted active item. Wraps Base UI ToggleGroup.
 *
 * Two non-obvious wrapper rules (see RESEARCH §Pattern 2 + Pitfalls 1-3):
 *
 * 1) Base UI's ToggleGroup `value` is `readonly Value[]` (array) even when
 *    `multiple={false}`. We adapt to a friendlier `value: string` /
 *    `onValueChange: (value: string) => void` API by wrapping in [value]
 *    and unwrapping next[0].
 *
 * 2) The correct prop is `multiple={false}` — NOT `toggleMultiple={false}`,
 *    which does not exist in Base UI v1.3. The UI-SPEC text was incorrect;
 *    the runtime contract is enforced here.
 *
 * 3) A single-select ToggleGroup will deselect the currently-active item
 *    if the user clicks it again, returning an empty array. For a
 *    P0/P1/P2/P3 picker (or Board/List toggle), an empty selection is
 *    invalid — there is no "no value" state. We swallow the empty-array
 *    deselect so a value is always defined.
 */

interface SegmentedControlProps {
  value: string
  onValueChange: (value: string) => void
  "aria-label": string
  children: React.ReactNode
  className?: string
  disabled?: boolean
}

function SegmentedControl({
  value,
  onValueChange,
  children,
  className,
  disabled,
  ...props
}: SegmentedControlProps) {
  return (
    <ToggleGroupPrimitive
      value={[value]}
      onValueChange={(next) => {
        // Single-select; swallow deselect to keep one value always set.
        if (next.length > 0) onValueChange(next[0])
      }}
      multiple={false}
      orientation="horizontal"
      disabled={disabled}
      data-slot="segmented-control"
      className={cn(
        "inline-flex h-6 w-fit items-center gap-0 rounded-lg bg-muted p-[3px]",
        className,
      )}
      {...props}
    >
      {children}
    </ToggleGroupPrimitive>
  )
}

interface SegmentedControlItemProps {
  value: string
  children: React.ReactNode
  disabled?: boolean
  className?: string
}

function SegmentedControlItem({
  value,
  children,
  disabled,
  className,
}: SegmentedControlItemProps) {
  return (
    <TogglePrimitive
      value={value}
      disabled={disabled}
      data-slot="segmented-control-item"
      className={cn(
        "inline-flex h-full min-w-8 items-center justify-center rounded-md px-3 text-sm font-medium text-muted-foreground transition-all",
        "hover:text-foreground",
        "focus-visible:ring-[3px] focus-visible:ring-ring/50 focus-visible:outline-none",
        "data-[pressed]:bg-background data-[pressed]:text-foreground data-[pressed]:shadow-sm",
        "disabled:pointer-events-none disabled:opacity-50",
        className,
      )}
    >
      {children}
    </TogglePrimitive>
  )
}

export { SegmentedControl, SegmentedControlItem }
export type { SegmentedControlProps, SegmentedControlItemProps }
```
